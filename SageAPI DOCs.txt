    //                              Power

    sage.power                          // system power managemnt

    # funcs
    sage.power.shutdown                 // shuts down the system
    sage.power.reboot                   // reboots the system



    //                              Memory


    sage.memory                         // memory managment

    # funcs
    sage.memory.malloc                  // allocates memory just like in C langauge
    sage.memory.free                    // frees the memory


    //                              Display

    sage.display                        // display managment

    # funcs
    sage.display.erase                  // erases one char
    sage.display.print                  // print only for strings (no pointers)
    sage.display.printf                 // printf just like in C langauge
    sage.display.drawLine               // draws line horizontaly
    sage.display.clean                  // clears the screen

    sage.display.changeFG               // changes foreground color (text color)
    sage.display.changeBG               // changes background color 

    sage.display.getFGcolor             // returns foreground color in hex
    sage.display.getBGcolor             // returns background color in hex

    sage.display.getColorName           // converts hex color to name
    sage.display.getColorHex            // converts color name to hex

    sage.display.resetFG                // resets foreground color
    sage.display.resetFBG               // resets background color               
    sage.display.resetFBG               // resets foreground and background color



    //                             Cursor

    sage.cursor                        // cursor managment

    # funcs
    sage.cursor.getCursorIndex         // returns cursor index
    sage.cursor.update                 // update cursor POS
